**2. Add PUT /api/jobs/:id/mockup-sent endpoint to a route file:**

```javascript
// ============================================================
// PUT /api/jobs/:id/mockup-sent — Mark mockup as sent (MOCKUP-05)
// ============================================================
router.put("/jobs/:id/mockup-sent", requireAuth, async (req, res, next) => {
  try {
    const result = await pool.query(
      "UPDATE jobs SET mockup_sent = true WHERE id = $1 RETURNING id, mockup_sent",
      [req.params.id]
    );

    if (result.rowCount === 0) {
      return res.status(404).json({ error: "Job not found" });
    }

    res.json({ message: "Mockup marked as sent", mockupSent: true });
  } catch (err) {
    next(err);
  }
});
```
