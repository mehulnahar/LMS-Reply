Check if a `PUT /api/jobs/:id` endpoint already exists. If not, add this before `module.exports = router`:

```javascript
// ============================================================
// PUT /api/jobs/:id/client-proposal-toggle — THREAD-03
// Persists the client_requested_proposal flag per lead
// ============================================================
router.put('/:id/client-proposal-toggle', requireAuth, async (req, res, next) => {
  try {
    const { clientRequestedProposal } = req.body;
    if (typeof clientRequestedProposal !== 'boolean') {
      return res.status(400).json({ error: 'clientRequestedProposal must be a boolean' });
    }
    const { rows } = await pool.query(
      'UPDATE jobs SET client_requested_proposal = $1 WHERE id = $2 AND user_id = $3 RETURNING id, client_requested_proposal',
      [clientRequestedProposal, req.params.id, req.user.id]
    );
    if (rows.length === 0) {
      return res.status(404).json({ error: 'Job not found' });
    }
    res.json({ id: rows[0].id, clientRequestedProposal: rows[0].client_requested_proposal });
  } catch (err) {
    next(err);
  }
});
```

**Part B: client/src/api.js — Add two API helper functions**

Read `client/src/api.js` first to find the existing pattern (likely named exports). Add these two functions following the same style:

```javascript
// THREAD-07: Increment email open count (manual hot signal tracking)
export const incrementOpenCount = async (emailId) => {
  const res = await fetch(`/api/emails/${emailId}/open-count`, {
    method: 'POST',
    credentials: 'include',
  });
  if (!res.ok) throw new Error('Failed to increment open count');
  return res.json();
};

// THREAD-03: Toggle client_requested_proposal flag on job (post-call recap mode)
export const togglePostCallRecap = async (jobId, clientRequestedProposal) => {
  const res = await fetch(`/api/jobs/${jobId}/client-proposal-toggle`, {
    method: 'PUT',
    credentials: 'include',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ clientRequestedProposal }),
  });
  if (!res.ok) throw new Error('Failed to update post-call proposal toggle');
  return res.json();
};
```
